CREATE TABLE rpt_definitions
(
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    name VARCHAR(255) NOT NULL
        COMMENT 'Ten hien thi',

    direction TINYINT UNSIGNED NOT NULL DEFAULT 0
        COMMENT '0: input, 1: output',

    node_type TINYINT UNSIGNED NOT NULL DEFAULT 0
        COMMENT '0: group - nhom bao cao; 1: file input/output thuc te',

    source_mode TINYINT UNSIGNED NOT NULL DEFAULT 0
        COMMENT '0: import - user upload; 1: generate - he thong sinh',

    file_kind TINYINT UNSIGNED NOT NULL DEFAULT 0
        COMMENT '0: NONE, 1: WORD, 2: EXCEL',

    allowed_extension VARCHAR(20) NOT NULL
        COMMENT 'Extension cho phep, vi du docx hoac xlsx',

    template_relative_path VARCHAR(1000) NULL
        COMMENT 'Duong dan tuong doi toi template; chu yeu dung cho output GENERATED',

    sort_order INT NOT NULL DEFAULT 0
        COMMENT 'Thu tu hien thi',

    status TINYINT NOT NULL DEFAULT 1
        COMMENT '1: dang su dung; 0: khong su dung',

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
		COMMENT 'Thoi gian tao',

    created_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi tao',

    PRIMARY KEY (id)

) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci
  COMMENT = 'Master dinh nghia nhom bao cao, file input và file output';

CREATE TABLE rpt_excel_header_defs (

CREATE TABLE rpt_section_defs
(
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    source_definition_id INT UNSIGNED NOT NULL
        COMMENT 'Definition cua file Word nguon tham chieu den bang rpt_definitions',

    section_name VARCHAR(255) NOT NULL
        COMMENT 'Ten section nghiep vu',

    extract_mode TINYINT NOT NULL DEFAULT 0
        COMMENT '
        0: TEXT_RANGE - lay cac node tu start_marker den end_marker,
        1: WHOLE_DOCUMENT - lay toan bo document,
        2: PARAGRAPH_MARKER - lay dung paragraph chua start_marker,
        3: NEXT_PARAGRAPHS - lay N paragraph sau paragraph chua start_marker,
        4: NEXT_LIST_ITEMS - lay N list item sau paragraph chua start_marker
        ',

    start_marker VARCHAR(1000) NULL
        COMMENT 'Text dung de xac dinh paragraph moc bat dau',

    end_marker VARCHAR(1000) NULL
        COMMENT '
        TEXT_RANGE: marker ket thuc range;
        NEXT_PARAGRAPHS/NEXT_LIST_ITEMS: marker dung de gioi han tim kiem;
        NULL = khong gioi han
        ',

    item_count INT UNSIGNED NULL
        COMMENT '
        So luong paragraph/list item can lay.
        Chi dung cho NEXT_PARAGRAPHS va NEXT_LIST_ITEMS
        ',

    include_start TINYINT NOT NULL DEFAULT 1
        COMMENT '
        Chi dung cho TEXT_RANGE.
        1: lay ca node chua start_marker; 0: khong lay
        ',

    include_end TINYINT NOT NULL DEFAULT 0
        COMMENT '
        Chi dung cho TEXT_RANGE.
        1: lay ca node chua end_marker; 0: khong lay
        ',

    status TINYINT NOT NULL DEFAULT 1
        COMMENT '1: dang su dung; 0: khong su dung',

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        COMMENT 'Thoi gian tao',

    created_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi tao',

    PRIMARY KEY (id),

    UNIQUE KEY uk_rpt_section_definition_name
        (source_definition_id, section_name)

) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci
  COMMENT = 'Dinh nghia cach xac dinh section can lay truc tiep tu file Word nguon';

CREATE TABLE rpt_instances
(
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    definition_id INT UNSIGNED NOT NULL
        COMMENT 'ID cua bao cao/du lieu vao tham chieu den bang rpt_definitions',

	current_file_id BIGINT UNSIGNED NULL
		COMMENT 'File version hien tai dang duoc su dung mapping den bang rpt_files',

    week_start_date DATE NOT NULL
        COMMENT 'Ngay Thu Ba bat dau tuan bao cao; ngay ket thuc = +6 ngay',

    status TINYINT NOT NULL DEFAULT 0
        COMMENT '0: EMPTY, 1: PROCESSING, 2: COMPLETED, 3: REJECTED, 4: ERROR, 5: GENERATING, 6: STALE',

    rejected_reason TEXT NULL COMMENT 'Ly do tu choi hien tai',

    rejected_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi tu choi',

    rejected_at DATETIME NULL,

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        COMMENT 'Thoi gian tao',

    created_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi tao',

	updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
		ON UPDATE CURRENT_TIMESTAMP
        COMMENT 'Thoi gian cap nhat cuoi',

    updated_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi cap nhat cuoi',

    PRIMARY KEY (id),

    UNIQUE KEY uk_rpt_instance_definition_week
        (definition_id, week_start_date),

    KEY idx_rpt_instance_current_file (current_file_id)

) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci
  COMMENT = 'Instance cua definition theo tuan, duoc lazy-create khi có thao tac thuc te';


CREATE TABLE rpt_files
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    report_instance_id INT UNSIGNED NOT NULL
        COMMENT 'Instance so huu file tham chieu den bang rpt_instances',

	version_no INT UNSIGNED NOT NULL
        COMMENT 'Lan import thanh cong: 1 = import lan dau, 2 tro len = re-import',

    original_filename VARCHAR(500) NOT NULL
        COMMENT 'Ten file hien thi/download; voi output la ten file duoc he thong sinh',

    storage_object_name VARCHAR(1000) NOT NULL
        COMMENT 'MinIO object name day du trong bucket, bao gom ten object vat ly',

    file_size BIGINT UNSIGNED NOT NULL DEFAULT 0
        COMMENT 'Dung lượng byte',

    sha256 CHAR(64)
        CHARACTER SET ascii
        COLLATE ascii_general_ci
        NOT NULL
        COMMENT 'SHA-256 cua noi dung file version',

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        COMMENT 'Thoi gian tao',

    created_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi tao',

	updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
		ON UPDATE CURRENT_TIMESTAMP
        COMMENT 'Thoi gian cap nhat cuoi',

    updated_by VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi cap nhat cuoi',

    PRIMARY KEY (id),

    UNIQUE KEY uk_rpt_file_instance_version
        (report_instance_id, version_no),

    KEY idx_rpt_file_instance_created
        (report_instance_id, created_at)

) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci
  COMMENT = 'Lich su cac version file cua report instance; moi version la bat bien';


CREATE TABLE rpt_audit_logs
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    week_start_date DATE NULL
        COMMENT 'Tuan lien quan',

    report_instance_id INT UNSIGNED NULL
        COMMENT 'Instance lien quan tham chieu den bang rpt_instances',

	file_id BIGINT UNSIGNED NULL
        COMMENT 'ID cua file mapping den bang rpt_files',

    reason TEXT NULL
		COMMENT 'Ly do cua thao tac, chu yeu dung cho REJECT',

    action TINYINT NOT NULL DEFAULT 0
        COMMENT '0: IMPORT, 1: REJECT, 2: DELETE',

    actor VARCHAR(100) NOT NULL DEFAULT ''
        COMMENT 'Nguoi tac dong',

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        COMMENT 'Thoi gian tao',

    PRIMARY KEY (id)

) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_unicode_ci
  COMMENT = 'Audit lich su thao tac import, reject, delete';
